**4. Aggregate revenue, units, marketing spend for Beauty Mobile App North before and after 2024-12-25**

```sql
WITH agg AS (
  SELECT order_date,
         CASE WHEN order_date < '2024-12-25' THEN 'pre' ELSE 'post' END AS period,
         gross_revenue::numeric AS rev,
         units,
         marketing_spend_usd::numeric AS spend
  FROM fct_daily_revenue
  WHERE category='Beauty' AND channel='Mobile App' AND region='North' AND order_date BETWEEN '2024-12-18' AND '2025-01-01'
)
SELECT period,
       COUNT(*) AS days,
       ROUND(AVG(rev),2) AS avg_rev,
       ROUND(SUM(rev),2) AS total_rev,
       ROUND(AVG(spend),2) AS avg_spend,
       ROUND(SUM(spend),2) AS total_spend,
       ROUND(AVG(units),2) AS avg_units,
       SUM(units) AS total_units
FROM agg
GROUP BY period
```
