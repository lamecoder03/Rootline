**6. Daily stockout rate for Electronics West during pre, anomaly, post windows**

```sql
WITH periods AS (
  SELECT order_date,
    CASE 
      WHEN order_date BETWEEN '2025-05-20' AND '2025-06-09' THEN 'pre'
      WHEN order_date BETWEEN '2025-06-10' AND '2025-06-16' THEN 'anomaly'
      WHEN order_date BETWEEN '2025-06-17' AND '2025-07-01' THEN 'post'
    END AS period
  FROM analytics.fct_daily_stockout
  WHERE category='Electronics' AND region='West' AND order_date BETWEEN '2025-05-20' AND '2025-07-01'
)
SELECT period, AVG(stockout_rate_pct) AS avg_stockout_rate, SUM(CASE WHEN stockout_rate_pct>0 THEN 1 ELSE 0 END) AS days_with_stockout
FROM analytics.fct_daily_stockout s
JOIN periods p ON s.order_date = p.order_date
GROUP BY period ORDER BY period;
```
